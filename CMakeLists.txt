  foreach(binding IN LISTS GBCLLR_CARTRIDGE_SYMBOLS)
    if(NOT binding MATCHES "^([0-9a-fA-F]+)=(.+)$")
      message(FATAL_ERROR "GBCLLR_CARTRIDGE_SYMBOLS requires SHA256=PATH bindings")
    endif()
    set(symbol_rom "${CMAKE_MATCH_1}")
    set(symbol_path "${CMAKE_MATCH_2}")
    string(TOLOWER "${symbol_rom}" symbol_rom)
    list(APPEND cartridge_symbol_roms "${symbol_rom}")
    list(APPEND extra_args --cartridge-symbols "${binding}")
    list(APPEND extra_deps "${symbol_path}")
  endforeach()

    list(LENGTH GBCLLR_ADDITIONAL_ROMS additional_count)

      list(APPEND program_byproducts
        ${program_dir}/cart_${index}/llr_program.inc
        ${program_dir}/cart_${index}/identity.h
        ${program_dir}/cart_${index}/program.json)
    endforeach()
  endif()
  foreach(pair SYMBOLS TRACE)
    if(GBCLLR_${pair})
      string(TOLOWER ${pair} flag)
      list(APPEND extra_args --${flag} ${GBCLLR_${pair}})
      list(APPEND extra_deps ${GBCLLR_${pair}})
    endif()
  endforeach()
  add_custom_command(OUTPUT ${program_dir}/llr_program.inc ${program_dir}/identity.h
    BYPRODUCTS ${program_byproducts}
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/recompile.py
      --rom ${GBCLLR_ROM} --output ${program_dir} ${extra_args}
    DEPENDS tools/recompile.py tools/native_blocks.py ${GBCLLR_ROM} ${extra_deps} ${SAMEBOY}/sm83_cpu.c
